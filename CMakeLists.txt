cmake_minimum_required(VERSION 3.16)
project(YMPME VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(YMPME_BUILD_TESTS "Build headless module tests" ON)
option(YMPME_BUILD_DEMOS "Build basic and game-loop demos" ON)
option(YMPME_BUILD_GRE_BRIDGE "Build optional YMGRE adapter" OFF)
option(YMPME_BUILD_GRE_DEMO "Build YMGRE bouncing-ball demo" OFF)
option(YMPME_GRE_WINDOW "Enable YMGUI/SDL window for the GRE demo" ON)
option(YMPME_ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
option(YMPME_BUILD_ROPE_GRE "Build rope GRE adapter without requiring demos or SDL" OFF)
option(YMPME_ENABLE_RIGID "Build optional 3D rigid-body dynamics" OFF)
option(YMPME_ENABLE_DYNAMICS "Build unified rigid/joint/contact step" OFF)
option(YMPME_ENABLE_JOINT "Build optional fixed/revolute/prismatic joints" OFF)
option(YMPME_MODEL_SURFACE_ATLAS "Build equal-area sensor display atlas" OFF)
option(YMPME_MODEL_SKIN "Build sampled compliant skin contact" OFF)
option(YMPME_BUILD_HAND_BALL_DEMO "Build independent hand ball playground" OFF)
option(YMPME_BUILD_HAND_SKIN_DEMO "Build low-poly hand tactile demo using YMSKE" OFF)
set(YMPME_SKE_ROOT "" CACHE PATH "YMSKE source root")
option(YMPME_MODEL_PENDULUM "Build optional reusable motor pendulum" OFF)
option(YMPME_ENABLE_QUERY "Build optional exact world ray queries" OFF)
option(YMPME_SENSOR_LASER "Build optional single-point laser range sensor" OFF)
option(YMPME_SENSOR_LIDAR "Build optional LiDAR sensor" OFF)
option(YMPME_SENSOR_DEPTH "Build optional depth camera sensor" OFF)
option(YMPME_SENSOR_TACTILE "Build optional tactile array sensor" OFF)
option(YMPME_SENSOR_IMU "Build optional IMU sensor" OFF)
option(YMPME_SENSOR_ENCODER "Build optional encoder sensor" OFF)
option(YMPME_SENSOR_FORCE "Build optional force sensor" OFF)
option(YMPME_MODEL_AIRCRAFT "Build optional drone/eagle flight model" OFF)
option(YMPME_MODEL_ROPE "Build optional independent rope model" OFF)
option(YMPME_MODEL_DRIVE "Build optional drive model" OFF)
option(YMPME_MODEL_TERRAIN "Build optional terrain model" OFF)
option(YMPME_MODEL_SUSPENSION "Build optional suspension model" OFF)
option(YMPME_MODEL_VEHICLE "Build optional vehicle model" OFF)
option(YMPME_MODEL_GEAR "Build optional gear model" OFF)
option(YMPME_MODEL_BELT "Build optional belt model" OFF)
option(YMPME_MODEL_PULLEY "Build optional pulley model" OFF)
option(YMPME_ENABLE_FLIGHT "Build optional flight load library" OFF)
option(YMPME_FLIGHT_ROTORS "Include rotor loads when flight is enabled" ON)
option(YMPME_FLIGHT_AERODYNAMICS "Include aerodynamic loads when flight is enabled" ON)
set(YMPME_GRE_ROOT "" CACHE PATH "YMGRE repository root")
if(YMPME_ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
	add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
	add_link_options(-fsanitize=address,undefined)
endif()
add_library(ympme STATIC
	YMPME/CONFIG/YMPME_Mem.c
	YMPME/OPOBJ/YMPME_Body.c
	YMPME/OPOBJ/YMPME_Creat.c
	YMPME/OPOBJ/YMPME_Free.c
	YMPME/CORE/YMPME_Collision.c
	YMPME/CORE/YMPME_ConvexMesh.c
	YMPME/CORE/YMPME_MeshSAT.c
	YMPME/CORE/YMPME_Solver.c
	YMPME/CORE/YMPME_World.c
	YMPME/DEBUG/YMPME_Debug.c)
add_library(YMPME::ympme ALIAS ympme)
target_include_directories(ympme PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/YMPME>
    $<INSTALL_INTERFACE:include/YMPME>)
if(NOT MSVC)
	target_link_libraries(ympme PUBLIC m)
	target_compile_options(ympme PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(YMPME_BUILD_DEMOS)
	add_executable(demo_free_fall Demo/demo_free_fall.c)
	target_link_libraries(demo_free_fall PRIVATE ympme)
	add_subdirectory(project_Demo/bouncing_ball)
endif()
if(YMPME_BUILD_TESTS)
	enable_testing()
	add_test(NAME ympme_demo_host_boundary COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_demo_host_boundary.cmake)
	add_executable(test_features_core tests/test_features_core.c)
	target_link_libraries(test_features_core PRIVATE ympme)
	add_test(NAME ympme_features_core COMMAND test_features_core)
	add_executable(test_mesh_collision tests/test_mesh_collision.c)
	target_link_libraries(test_mesh_collision PRIVATE ympme)
	add_test(NAME ympme_mesh_collision COMMAND test_mesh_collision)
	add_executable(test_physics tests/test_physics.c)
	target_link_libraries(test_physics PRIVATE ympme)
	foreach(case integration collision response resting lifecycle fixed_step filtering memory)
		add_test(NAME ympme_${case} COMMAND test_physics ${case})
	endforeach()
endif()
if(YMPME_MODEL_VEHICLE AND NOT (YMPME_MODEL_DRIVE AND YMPME_MODEL_TERRAIN AND YMPME_MODEL_SUSPENSION))
 message(FATAL_ERROR "VEHICLE requires YMPME_MODEL_DRIVE, YMPME_MODEL_TERRAIN and YMPME_MODEL_SUSPENSION")
endif()
foreach(model DRIVE TERRAIN SUSPENSION VEHICLE GEAR BELT PULLEY)
 if(YMPME_MODEL_${model})
  add_subdirectory(YMPME/MODEL/${model})
 endif()
endforeach()
if(YMPME_MODEL_ROPE)
	add_subdirectory(YMPME/MODEL/ROPE)
endif()
if(YMPME_BUILD_ROPE_GRE AND NOT YMPME_MODEL_ROPE)
    message(FATAL_ERROR "YMPME_BUILD_ROPE_GRE requires YMPME_MODEL_ROPE=ON")
endif()
if(YMPME_BUILD_ROPE_GRE AND NOT EXISTS "${YMPME_GRE_ROOT}/YMGRE/OPOBJ/YMGRE_OBJ.h")
    message(FATAL_ERROR "Set YMPME_GRE_ROOT to the YMGRE repository root")
endif()
if(YMPME_BUILD_GRE_BRIDGE OR YMPME_BUILD_GRE_DEMO)
	if(NOT EXISTS "${YMPME_GRE_ROOT}/YMGRE/OPOBJ/YMGRE_OBJ.h")
		message(FATAL_ERROR "Set YMPME_GRE_ROOT to the YMGRE repository root")
	endif()
	add_library(ympme_gre STATIC YMPME/BRIDGE/YMPME_GRE.c)
	target_include_directories(ympme_gre PUBLIC ${YMPME_GRE_ROOT})
	target_link_libraries(ympme_gre PUBLIC ympme)
	if(YMPME_BUILD_TESTS)
		add_executable(test_gre_bridge tests/test_gre_bridge.c)
		target_link_libraries(test_gre_bridge PRIVATE ympme_gre)
		add_test(NAME ympme_gre_bridge COMMAND test_gre_bridge)
	endif()
endif()
if(YMPME_BUILD_GRE_DEMO OR YMPME_BUILD_ROPE_GRE)
	if(NOT TARGET ymgre)
		set(YMGRE_BUILD_DEMOS OFF)
		set(YMGRE_BUILD_TESTS OFF)
		set(YMGRE_BUILD_YMGUI_HOST OFF)
		if(YMPME_BUILD_GRE_DEMO AND YMPME_GRE_WINDOW)
			set(YMGRE_BUILD_YMGUI_HOST ON)
		endif()
		add_subdirectory("${YMPME_GRE_ROOT}" "extern_lib/YMGRE" EXCLUDE_FROM_ALL)
	endif()
	# 与宿主 GRE 的公开编译定义保持一致（包括 framebuffer 色深）。
	if(TARGET ympme_gre)
		target_link_libraries(ympme_gre PUBLIC ymgre)
	endif()
	if(YMPME_MODEL_ROPE)
		add_library(ympme_rope_gre STATIC YMPME/BRIDGE/YMPME_RopeGRE.c)
		add_library(YMPME::rope_gre ALIAS ympme_rope_gre)
		target_include_directories(ympme_rope_gre PUBLIC ${YMPME_GRE_ROOT})
		target_link_libraries(ympme_rope_gre PUBLIC YMPME::rope ymgre)
		if(YMPME_BUILD_TESTS)
			add_executable(test_rope_gre tests/test_rope_gre.c)
			target_link_libraries(test_rope_gre PRIVATE YMPME::rope_gre)
			add_test(NAME ympme_rope_gre COMMAND test_rope_gre)
			if(UNIX AND NOT APPLE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
				target_compile_definitions(test_rope_gre PRIVATE PME_TEST_ALLOC_WRAP=1)
				target_link_options(test_rope_gre PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc)
			endif()
		endif()
	endif()
	if(YMPME_BUILD_GRE_DEMO)
		add_subdirectory(project_Demo/gre_bouncing_ball)
		if(YMPME_GRE_WINDOW)
			if(YMPME_MODEL_DRIVE)
				add_subdirectory(project_Demo/three_wheel_car)
			endif()
			if(YMPME_MODEL_VEHICLE)
				add_subdirectory(project_Demo/vehicle_playground)
			endif()
			if(YMPME_MODEL_GEAR)
				add_subdirectory(project_Demo/gear_mesh)
			endif()
			if(YMPME_MODEL_ROPE)
				if(YMPME_MODEL_BELT AND YMPME_MODEL_PULLEY)
					add_subdirectory(project_Demo/mechanisms)
				endif()
				add_subdirectory(project_Demo/rope_lab)
			else()
				message(STATUS "Enable YMPME_MODEL_ROPE for mechanisms and rope_lab demos")
			endif()
		endif()
	endif()

endif()

	add_subdirectory(YMPME/EXT/FLIGHT)

if(YMPME_SENSOR_LASER OR YMPME_SENSOR_IMU OR YMPME_SENSOR_ENCODER OR YMPME_SENSOR_FORCE OR YMPME_SENSOR_LIDAR OR YMPME_SENSOR_DEPTH OR YMPME_SENSOR_TACTILE)

 add_executable(test_sensor_scene tests/test_sensor_scene.c)
 target_link_libraries(test_sensor_scene PRIVATE YMPME::query YMPME::lidar YMPME::depth YMPME::tactile)
 add_test(NAME ympme_sensor_scene COMMAND test_sensor_scene)

 add_subdirectory(YMPME/MODEL/SURFACE_ATLAS)

 add_subdirectory(YMPME/MODEL/SKIN)
