cmake_minimum_required(VERSION 3.16)
project(YMPME_Owl_Flight LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(PME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(YMPME_GRE_ROOT "${PME_ROOT}/../0502_YMGRE" CACHE PATH "GRE source")
set(YMPME_SKE_ROOT "${PME_ROOT}/../YMSKE" CACHE PATH "SKE source")
set(YMPME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMPME_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMPME_MODEL_AIRCRAFT ON CACHE BOOL "" FORCE)
set(YMPME_ENABLE_FLIGHT ON CACHE BOOL "" FORCE)
add_subdirectory("${PME_ROOT}" pme EXCLUDE_FROM_ALL)
set(YMGRE_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_YMGUI_HOST ON CACHE BOOL "" FORCE)
add_subdirectory("${YMPME_GRE_ROOT}" gre EXCLUDE_FROM_ALL)
set(YMSKE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_HAND_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_OWL_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_PME OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_SKIN ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_IO ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_GRE ON CACHE BOOL "" FORCE)
set(YMSKE_MAX_INFLUENCES 4 CACHE STRING "Owl uses four weights" FORCE)
set(YMSKE_GRE_ROOT "${YMPME_GRE_ROOT}" CACHE PATH "" FORCE)
add_subdirectory("${YMPME_SKE_ROOT}" ske EXCLUDE_FROM_ALL)
add_library(owl_flight_assets STATIC "${YMPME_SKE_ROOT}/project_Demo/owl/scene.c")
target_compile_definitions(owl_flight_assets PRIVATE OWL_ASSET_ROOT="${YMPME_SKE_ROOT}/Resource/owl")
target_include_directories(owl_flight_assets PUBLIC "${YMPME_SKE_ROOT}/project_Demo")
target_link_libraries(owl_flight_assets PUBLIC YMSKE::gre YMSKE::io)
add_library(owl_flight_files OBJECT "${YMPME_GRE_ROOT}/YMGRE/CORE/YMGRE_SceneManager.c")
target_link_libraries(owl_flight_files PRIVATE ymgre)
add_library(owl_flight_scene STATIC flight.c)
target_sources(owl_flight_scene INTERFACE $<TARGET_OBJECTS:owl_flight_files>)
target_link_libraries(owl_flight_scene PUBLIC owl_flight_assets YMPME::aircraft)
add_executable(demo_owl_flight main.c)
target_link_libraries(demo_owl_flight PRIVATE owl_flight_scene ymgre_demo_host)
set(OWL_FLIGHT_OUTPUT_ROOT "${PME_ROOT}/build/project_Demo" CACHE PATH "Demo executable root")
set_target_properties(demo_owl_flight PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OWL_FLIGHT_OUTPUT_ROOT}/owl_flight")
add_executable(test_owl_flight test_flight.c)
target_link_libraries(test_owl_flight PRIVATE owl_flight_scene)
if(NOT MSVC)
 foreach(t owl_flight_scene demo_owl_flight test_owl_flight)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
 endforeach()
endif()
if(UNIX AND NOT APPLE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_definitions(test_owl_flight PRIVATE WRAP_ALLOC=1)
 target_link_options(test_owl_flight PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
endif()
enable_testing()
add_test(NAME owl_flight_dynamics COMMAND test_owl_flight)
add_test(NAME owl_flight_window COMMAND demo_owl_flight --smoke-test)
set_tests_properties(owl_flight_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0;YMGRE_WINDOW_SCALE=1" TIMEOUT 90)
set_tests_properties(owl_flight_dynamics PROPERTIES TIMEOUT 90)
