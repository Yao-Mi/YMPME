cmake_minimum_required(VERSION 3.16)
project(YMPME_Human_Skin LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(PME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(YMPME_GRE_ROOT "${PME_ROOT}/../0502_YMGRE" CACHE PATH "GRE source")
set(YMPME_SKE_ROOT "${PME_ROOT}/../YMSKE" CACHE PATH "SKE source")
option(HUMAN_SKIN_SANITIZERS "Address and undefined behavior checks" OFF)
if(HUMAN_SKIN_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
# 六槽人体使用独立构建/链接图，不改变已有四槽hand demo的缓存或ABI。
set(YMPME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMPME_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMPME_MODEL_SURFACE_ATLAS ON CACHE BOOL "" FORCE)
set(YMPME_MODEL_SKIN ON CACHE BOOL "" FORCE)
set(YMPME_SENSOR_TACTILE ON CACHE BOOL "" FORCE)
set(YMPME_SENSOR_FORCE ON CACHE BOOL "" FORCE)
add_subdirectory("${PME_ROOT}" pme EXCLUDE_FROM_ALL)
set(YMGRE_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMGRE_BUILD_YMGUI_HOST ON CACHE BOOL "" FORCE)
add_subdirectory("${YMPME_GRE_ROOT}" gre EXCLUDE_FROM_ALL)
set(YMSKE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_HAND_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_OWL_DEMO OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_PME OFF CACHE BOOL "" FORCE)
set(YMSKE_BUILD_SKIN ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_IO ON CACHE BOOL "" FORCE)
set(YMSKE_BUILD_GRE ON CACHE BOOL "" FORCE)
set(YMSKE_MAX_INFLUENCES 6 CACHE STRING "Human needs all six influence slots" FORCE)
set(YMSKE_GRE_ROOT "${YMPME_GRE_ROOT}" CACHE PATH "" FORCE)
add_subdirectory("${YMPME_SKE_ROOT}" ske EXCLUDE_FROM_ALL)
add_library(human_skin_atlas STATIC "${YMPME_GRE_ROOT}/project_Demo/scene_baker/scene_uv_atlas.cpp" "${YMPME_GRE_ROOT}/extern_lib/xatlas/xatlas.cpp")
target_include_directories(human_skin_atlas PUBLIC "${YMPME_GRE_ROOT}/project_Demo/scene_baker" PRIVATE "${YMPME_GRE_ROOT}/extern_lib/xatlas")
target_compile_features(human_skin_atlas PRIVATE cxx_std_11)
target_compile_definitions(human_skin_atlas PRIVATE XA_MULTITHREADED=0)
add_library(human_skin_files OBJECT "${YMPME_GRE_ROOT}/YMGRE/CORE/YMGRE_SceneManager.c")
target_link_libraries(human_skin_files PRIVATE ymgre)
add_library(human_skin_scene STATIC rig.c self_touch.c ground.c observation.c contact_view.c uv.c ../common/force_channels.c "${YMPME_SKE_ROOT}/project_Demo/human/scene.c")
target_sources(human_skin_scene INTERFACE $<TARGET_OBJECTS:human_skin_files>)
target_include_directories(human_skin_scene PUBLIC . ../common "${YMPME_SKE_ROOT}/project_Demo/human")
target_compile_definitions(human_skin_scene PRIVATE HUMAN_ASSET_ROOT="${YMPME_SKE_ROOT}/Resource/human")
target_link_libraries(human_skin_scene PUBLIC YMSKE::gre YMSKE::io YMPME::skin_contact YMPME::surface_atlas YMPME::tactile YMPME::force_sensor human_skin_atlas)
if(NOT MSVC)
 target_compile_options(human_skin_scene PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
endif()
add_executable(demo_human_skin main.c)
target_link_libraries(demo_human_skin PRIVATE human_skin_scene ymgre_demo_host)
if(NOT MSVC)
 target_compile_options(demo_human_skin PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
endif()
enable_testing()
add_test(NAME human_skin_window COMMAND demo_human_skin --smoke-test)
set_tests_properties(human_skin_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0" TIMEOUT 60)

add_executable(test_human_skin test_rig.c)
target_link_libraries(test_human_skin PRIVATE human_skin_scene)
if(UNIX AND NOT APPLE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_definitions(test_human_skin PRIVATE WRAP_ALLOC=1)
 target_link_options(test_human_skin PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
endif()
add_test(NAME human_skin_feedback COMMAND test_human_skin)
set_tests_properties(human_skin_feedback PROPERTIES TIMEOUT 120)

add_executable(test_human_self test_self.c)
target_link_libraries(test_human_self PRIVATE human_skin_scene)
if(UNIX AND NOT APPLE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_definitions(test_human_self PRIVATE WRAP_ALLOC=1)
 target_link_options(test_human_self PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
endif()
if(NOT MSVC)
 target_compile_options(test_human_self PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
endif()
add_test(NAME human_self_contact COMMAND test_human_self)
set_tests_properties(human_self_contact PROPERTIES TIMEOUT 120)
add_executable(test_skin_self_contact ../../tests/test_skin_self_contact.c)
target_link_libraries(test_skin_self_contact PRIVATE YMPME::skin_contact)
add_test(NAME skin_self_contact_core COMMAND test_skin_self_contact)
add_executable(test_surface_atlas ../../tests/test_surface_atlas.c)
target_link_libraries(test_surface_atlas PRIVATE YMPME::surface_atlas)
add_test(NAME surface_atlas_core COMMAND test_surface_atlas)

set(HUMAN_DEMO_OUTPUT_ROOT "${CMAKE_BINARY_DIR}/project_Demo" CACHE PATH "Human demo executable directory")
set_target_properties(demo_human_skin PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${HUMAN_DEMO_OUTPUT_ROOT}/human_skin")
add_executable(demo_human_ground ../human_ground/main.c)
target_link_libraries(demo_human_ground PRIVATE human_skin_scene ymgre_demo_host)
set_target_properties(demo_human_ground PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${HUMAN_DEMO_OUTPUT_ROOT}/human_ground")
add_test(NAME human_ground_window COMMAND demo_human_ground --smoke-test)
set_tests_properties(human_ground_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0" TIMEOUT 60)
add_executable(test_human_ground ../human_ground/test_ground.c)
target_link_libraries(test_human_ground PRIVATE human_skin_scene)
add_test(NAME human_ground_feedback COMMAND test_human_ground)
set_tests_properties(human_ground_feedback PROPERTIES TIMEOUT 120)
add_executable(test_skin_plane ../../tests/test_skin_plane.c)
target_link_libraries(test_skin_plane PRIVATE YMPME::skin_contact)
add_test(NAME skin_plane_core COMMAND test_skin_plane)
if(NOT MSVC)
 foreach(t demo_human_ground test_human_ground)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
 endforeach()
endif()

add_executable(test_topology_grid ../../tests/test_topology_grid.c)
target_link_libraries(test_topology_grid PRIVATE YMPME::surface_atlas)
add_test(NAME topology_grid_core COMMAND test_topology_grid)
