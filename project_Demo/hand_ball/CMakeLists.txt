# GRE编辑器的同一xatlas实现仅供此桌面demo初始化使用；PME/SKE核心仍为C99。
enable_language(CXX)
if(NOT EXISTS "${YMPME_GRE_ROOT}/project_Demo/scene_baker/scene_uv_atlas.cpp")
 message(FATAL_ERROR "Hand ball needs GRE scene_baker automatic UV adapter")
endif()
add_library(ympme_hand_uv_atlas STATIC
 "${YMPME_GRE_ROOT}/project_Demo/scene_baker/scene_uv_atlas.cpp"
 "${YMPME_GRE_ROOT}/extern_lib/xatlas/xatlas.cpp")
target_include_directories(ympme_hand_uv_atlas PUBLIC "${YMPME_GRE_ROOT}/project_Demo/scene_baker" PRIVATE "${YMPME_GRE_ROOT}/extern_lib/xatlas")
target_compile_features(ympme_hand_uv_atlas PRIVATE cxx_std_11)
target_compile_definitions(ympme_hand_uv_atlas PRIVATE XA_MULTITHREADED=0)
add_library(ympme_hand_ball STATIC ball.c force_view.c ../common/force_channels.c)
target_link_libraries(ympme_hand_ball PUBLIC ympme_hand_skin ympme_hand_uv_atlas)
target_include_directories(ympme_hand_ball PUBLIC .)
if(NOT MSVC)
 target_compile_options(ympme_hand_ball PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
endif()
add_executable(demo_hand_ball main.c)
target_link_libraries(demo_hand_ball PRIVATE ympme_hand_ball ymgre_demo_host)
if(NOT MSVC)
 target_compile_options(demo_hand_ball PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
endif()
if(YMPME_BUILD_TESTS)
 add_executable(test_hand_ball test_ball.c)
 target_link_libraries(test_hand_ball PRIVATE ympme_hand_ball)
 if(UNIX AND NOT APPLE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_definitions(test_hand_ball PRIVATE WRAP_ALLOC=1)
  target_link_options(test_hand_ball PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
 endif()
 add_test(NAME ympme_hand_ball_physics COMMAND test_hand_ball)
 add_test(NAME ympme_hand_ball_window COMMAND demo_hand_ball --smoke-test)
 set_tests_properties(ympme_hand_ball_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0" TIMEOUT 60)
endif()
