# The physics/model and its tests remain usable without any display backend.
add_library(ympme_grasp_lab STATIC grasp.c shape.c)
target_link_libraries(ympme_grasp_lab PUBLIC YMPME::dynamics YMPME::tactile)
if(NOT MSVC)
 target_compile_options(ympme_grasp_lab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(YMPME_BUILD_GRE_DEMO AND YMPME_GRE_WINDOW)
 add_executable(demo_grasp_lab main.c scene.c)
 target_link_libraries(demo_grasp_lab PRIVATE ympme_grasp_lab ymgre ymgre_demo_host)
 if(NOT MSVC)
  target_compile_options(demo_grasp_lab PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
 endif()
 if(YMPME_BUILD_TESTS)
  add_test(NAME ympme_grasp_window COMMAND demo_grasp_lab --smoke-test)
  set_tests_properties(ympme_grasp_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0;YMGRE_WINDOW_SCALE=1" TIMEOUT 90)
 endif()
endif()
if(YMPME_BUILD_TESTS)
 add_executable(test_grasp test_grasp.c)
 target_link_libraries(test_grasp PRIVATE ympme_grasp_lab)
 add_test(NAME ympme_grasp_physics COMMAND test_grasp)
 add_test(NAME ympme_grasp_warm_start COMMAND test_grasp --warm-start)
 add_test(NAME ympme_grasp_warm_all COMMAND test_grasp --warm-all)
endif()
