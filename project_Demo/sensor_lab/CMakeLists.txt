add_executable(demo_sensor_lab main.c rig.c scene.c lidar_projection.c)
target_link_libraries(demo_sensor_lab PRIVATE YMPME::query YMPME::lidar YMPME::depth YMPME::tactile ymgre ymgre_demo_host)
if(NOT MSVC)
 target_compile_options(demo_sensor_lab PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
endif()
if(YMPME_BUILD_TESTS)
 add_test(NAME ympme_sensor_window COMMAND demo_sensor_lab --smoke-test)
 set_tests_properties(ympme_sensor_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0;YMGRE_WINDOW_SCALE=1" TIMEOUT 60)
endif()

foreach(sensor IMU ENCODER FORCE LASER)
 if(YMPME_SENSOR_${sensor})
  if(sensor STREQUAL "FORCE")
   target_link_libraries(demo_sensor_lab PRIVATE YMPME::force_sensor)
  else()
   string(TOLOWER "${sensor}" name)
   target_link_libraries(demo_sensor_lab PRIVATE YMPME::${name})
  endif()
 endif()
endforeach()
