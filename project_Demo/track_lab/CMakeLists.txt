# Standalone use is useful for an isolated physics-only build.
cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(YMPMETrackLab C)
 set(CMAKE_C_STANDARD 99)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(YMPME_TRACK_STANDALONE ON)
 set(YMPME_ENABLE_DYNAMICS ON CACHE BOOL "" FORCE)
 set(YMPME_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
 set(YMPME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 if(NOT YMPME_SOURCE_ROOT)
  get_filename_component(YMPME_SOURCE_ROOT "../.." ABSOLUTE)
 endif()
 add_subdirectory("${YMPME_SOURCE_ROOT}" "pme" EXCLUDE_FROM_ALL)
endif()
add_library(ympme_track_bench STATIC track.c)
target_link_libraries(ympme_track_bench PUBLIC YMPME::dynamics)
if(NOT MSVC)
 target_compile_options(ympme_track_bench PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(YMPME_BUILD_TESTS OR YMPME_TRACK_STANDALONE)
 add_executable(test_track_lab test_track.c)
 target_link_libraries(test_track_lab PRIVATE ympme_track_bench)
endif()
if(YMPME_BUILD_TESTS)
 add_test(NAME ympme_track_bench COMMAND test_track_lab)
 add_test(NAME ympme_track_warm_start COMMAND test_track_lab --warm-start)
 add_test(NAME ympme_track_warm_all COMMAND test_track_lab --warm-all)
endif()
if(TARGET ymgre_demo_host)
 add_executable(demo_track_lab main.c scene.c)
 target_link_libraries(demo_track_lab PRIVATE ympme_track_bench ymgre ymgre_demo_host)
 if(NOT MSVC)
  target_compile_options(demo_track_lab PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
 endif()
endif()

if(YMPME_BUILD_TESTS AND TARGET demo_track_lab)
 add_test(NAME ympme_track_window COMMAND demo_track_lab --smoke-test)
 set_tests_properties(ympme_track_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0;YMGRE_WINDOW_SCALE=1" TIMEOUT 90)
endif()
