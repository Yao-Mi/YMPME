cmake_minimum_required(VERSION 3.16)
project(YMPME_SoftFinger LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SOFT_FINGER_WINDOW "Build GRE interactive demo" ON)
option(SOFT_FINGER_SANITIZERS "Sanitize isolated soft-finger build" OFF)
if(SOFT_FINGER_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(YMPME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YMPME_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(YMPME_BUILD_GRE_DEMO ${SOFT_FINGER_WINDOW} CACHE BOOL "" FORCE)
set(YMPME_GRE_WINDOW ${SOFT_FINGER_WINDOW} CACHE BOOL "" FORCE)
set(YMPME_ENABLE_DYNAMICS ON CACHE BOOL "" FORCE)
set(YMPME_SENSOR_TACTILE ON CACHE BOOL "" FORCE)
set(YMPME_TRACK_STANDALONE ON)
set(YMPME_TRACKED_STANDALONE ON)
add_subdirectory(../.. pme EXCLUDE_FROM_ALL)
if(NOT TARGET YMPME::softbody)
 add_subdirectory(../../YMPME/SOFTBODY softbody)
endif()
if(NOT TARGET ympme_grasp_lab)
 add_subdirectory(../grasp_lab grasp EXCLUDE_FROM_ALL)
endif()
add_library(ympme_soft_finger STATIC soft_finger.c ball.c)
target_include_directories(ympme_soft_finger PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ympme_soft_finger PUBLIC ympme_grasp_lab YMPME::softbody)
if(NOT MSVC)
 target_compile_options(ympme_soft_finger PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(test_soft_ball test_soft_ball.c)
target_link_libraries(test_soft_ball PRIVATE YMPME::softbody)
add_test(NAME soft_ball_material COMMAND test_soft_ball)
add_executable(test_soft_pad test_soft_pad.c)
target_link_libraries(test_soft_pad PRIVATE YMPME::softbody)
add_test(NAME soft_pad_material COMMAND test_soft_pad)
add_executable(test_soft_finger test_soft_finger.c)
target_link_libraries(test_soft_finger PRIVATE ympme_soft_finger)
add_test(NAME soft_finger_grip COMMAND test_soft_finger)
if(SOFT_FINGER_WINDOW)
 add_executable(demo_soft_finger main.c render.c ../grasp_lab/scene.c)
 target_link_libraries(demo_soft_finger PRIVATE ympme_soft_finger ymgre ymgre_demo_host)
 set_target_properties(demo_soft_finger PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/project_Demo/soft_finger")
 if(NOT MSVC)
  target_compile_options(demo_soft_finger PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
 endif()
 add_test(NAME soft_finger_window COMMAND demo_soft_finger --smoke-test)
 set_tests_properties(soft_finger_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0" TIMEOUT 90)
endif()
