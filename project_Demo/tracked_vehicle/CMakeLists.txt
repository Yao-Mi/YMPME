cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(YMPMETrackedVehicle C)
 set(CMAKE_C_STANDARD 99)
 set(CMAKE_C_STANDARD_REQUIRED ON)
 set(YMPME_TRACKED_STANDALONE ON)
 option(TRACKED_WINDOW "Build GRE window" ON)
 set(YMPME_ENABLE_DYNAMICS ON CACHE BOOL "" FORCE)
 set(YMPME_MODEL_SUSPENSION ON CACHE BOOL "" FORCE)
 set(YMPME_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
 set(YMPME_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 if(NOT YMPME_SOURCE_ROOT)
  get_filename_component(YMPME_SOURCE_ROOT "../.." ABSOLUTE)
 endif()
 add_subdirectory("${YMPME_SOURCE_ROOT}" pme EXCLUDE_FROM_ALL)
 if(TRACKED_WINDOW AND NOT TARGET ymgre_demo_host)
  if(NOT EXISTS "${YMPME_GRE_ROOT}/CMakeLists.txt")
   message(FATAL_ERROR "Set YMPME_GRE_ROOT, or TRACKED_WINDOW=OFF for physics tests")
  endif()
  set(YMGRE_BUILD_DEMOS OFF)
  set(YMGRE_BUILD_TESTS OFF)
  set(YMGRE_BUILD_YMGUI_HOST ON)
  add_subdirectory("${YMPME_GRE_ROOT}" gre EXCLUDE_FROM_ALL)
 endif()
 enable_testing()
endif()
add_library(ympme_tracked_vehicle STATIC vehicle.c)
target_link_libraries(ympme_tracked_vehicle PUBLIC YMPME::dynamics YMPME::suspension)
if(NOT MSVC)
 target_compile_options(ympme_tracked_vehicle PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(YMPME_BUILD_TESTS OR YMPME_TRACKED_STANDALONE)
 add_executable(test_tracked_vehicle test_vehicle.c)
 target_link_libraries(test_tracked_vehicle PRIVATE ympme_tracked_vehicle)
 foreach(case forward reverse turn wall terrain friction)
  add_test(NAME ympme_tracked_vehicle_${case} COMMAND test_tracked_vehicle ${case})
  set_tests_properties(ympme_tracked_vehicle_${case} PROPERTIES TIMEOUT 180)
 endforeach()
endif()
if(TARGET ymgre_demo_host)
 add_executable(demo_tracked_vehicle main.c scene.c)
 target_link_libraries(demo_tracked_vehicle PRIVATE ympme_tracked_vehicle ymgre ymgre_demo_host)
 if(NOT MSVC)
  target_compile_options(demo_tracked_vehicle PRIVATE -Wall -Wextra -Werror -Wno-unknown-pragmas -Wno-unused-parameter -Wno-comment)
 endif()
 if(YMPME_BUILD_TESTS OR YMPME_TRACKED_STANDALONE)
  add_test(NAME ympme_tracked_vehicle_window COMMAND demo_tracked_vehicle --smoke-test)
  set_tests_properties(ympme_tracked_vehicle_window PROPERTIES ENVIRONMENT "YMGRE_HEADLESS=1;YMGRE_MAX_FRAMES=0;YMGRE_WINDOW_SCALE=1" TIMEOUT 180)
 endif()
endif()
