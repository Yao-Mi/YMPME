add_executable(test_debug test_debug.c)
target_link_libraries(test_debug PRIVATE ympme)
if(YMPME_ENABLE_DYNAMICS)
    # The dynamics target is defined later in the parent CMakeLists.
    target_link_libraries(test_debug PRIVATE ympme_dynamics)
endif()
add_test(NAME ympme_debug COMMAND test_debug)
add_executable(test_debug_disabled test_disabled.c ../../YMPME/DEBUG/YMPME_Log.c)
target_include_directories(test_debug_disabled PRIVATE ../../YMPME)
target_compile_definitions(test_debug_disabled PRIVATE PME_ENABLE_LOGGING=0 PME_DEBUG_STDERR=0 PME_ENABLE_ASSERTS=0)
add_test(NAME ympme_debug_disabled COMMAND test_debug_disabled)
foreach(mode asserts callback_only logging_off_asserts)
    add_executable(test_debug_${mode} test_modes.c ../../YMPME/DEBUG/YMPME_Log.c)
    target_include_directories(test_debug_${mode} PRIVATE ../../YMPME)
    if(mode STREQUAL "asserts")
        set(sink 1)
    else()
        set(sink 0)
    endif()
    if(mode STREQUAL "logging_off_asserts")
        set(logging 0)
    else()
        set(logging 1)
    endif()
    target_compile_definitions(test_debug_${mode} PRIVATE PME_ENABLE_LOGGING=${logging} PME_DEBUG_STDERR=${sink} PME_ENABLE_ASSERTS=1)
    add_test(NAME ympme_debug_${mode} COMMAND ${CMAKE_COMMAND}
        -DPROGRAM=$<TARGET_FILE:test_debug_${mode}> -DEXPECT_STDERR=${sink} -DEXPECT_ASSERT=1
        -P ${CMAKE_CURRENT_SOURCE_DIR}/check_modes.cmake)
endforeach()
add_executable(test_debug_defaults test_modes.c)
target_link_libraries(test_debug_defaults PRIVATE ympme)
if(YMPME_ENABLE_LOGGING AND YMPME_DEBUG_STDERR)
    set(default_sink 1)
else()
    set(default_sink 0)
endif()
add_test(NAME ympme_debug_defaults COMMAND ${CMAKE_COMMAND}
    -DPROGRAM=$<TARGET_FILE:test_debug_defaults> -DEXPECT_STDERR=${default_sink}
    -DEXPECT_ASSERT=${YMPME_ENABLE_ASSERTS} -P ${CMAKE_CURRENT_SOURCE_DIR}/check_modes.cmake)
foreach(target test_debug test_debug_disabled test_debug_asserts test_debug_callback_only test_debug_logging_off_asserts test_debug_defaults)
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
