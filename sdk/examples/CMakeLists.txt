cmake_minimum_required(VERSION 3.16)
project(YMPMESDKExamples C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# This folder also works when copied out of the SDK; pass YMPME_DIR in that case.
find_package(YMPME CONFIG REQUIRED HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../cmake")
enable_testing()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
function(sdk_example name)
    add_executable(${name} ${name}.c)
    if(YMPME_SDK_UNIFIED)
        target_link_libraries(${name} PRIVATE YMPME::ympme ${ARGN})
    else()
        target_link_libraries(${name} PRIVATE ${ARGN})
    endif()
    if(NOT MSVC)
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    install(TARGETS ${name} RUNTIME DESTINATION bin)
endfunction()
sdk_example(demo_sdk_minimal YMPME::ympme)
sdk_example(demo_sdk_debug YMPME::ympme)
if(TARGET YMPME::dynamics)
    sdk_example(demo_sdk_dynamics YMPME::dynamics)
endif()
if(TARGET YMPME::encoder)
    sdk_example(demo_sdk_encoder YMPME::encoder)
endif()

set(_mesh_targets YMPME::ympme)
if(TARGET YMPME::dynamics)
    list(APPEND _mesh_targets YMPME::dynamics)
endif()
if(TARGET YMPME::query)
    list(APPEND _mesh_targets YMPME::query)
endif()
sdk_example(demo_sdk_mesh_indices ${_mesh_targets})
option(YMPME_SDK_WITH_GRE "Build native shared-geometry example against an external GRE SDK" OFF)
if(YMPME_SDK_WITH_GRE)
    if(DEFINED YMGRE_INDEX_BITS AND NOT YMGRE_INDEX_BITS STREQUAL "${YMPME_INDEX_BITS}")
        message(FATAL_ERROR "Shared PME/GRE geometry requires matching index widths")
    endif()
    set(YMGRE_INDEX_BITS ${YMPME_INDEX_BITS})
    find_package(YMGRE CONFIG REQUIRED)
    add_library(ympme_shared_gre STATIC gre/YMPME_GREMesh.c)
    target_include_directories(ympme_shared_gre PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/gre")
    target_link_libraries(ympme_shared_gre PUBLIC YMPME::ympme YMGRE::ymgre)
    sdk_example(demo_sdk_shared_gre ympme_shared_gre)
endif()
