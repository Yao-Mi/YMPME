add_library(ympme_flight STATIC YMPME_Flight.c)
add_library(YMPME::flight ALIAS ympme_flight)
target_link_libraries(ympme_flight PUBLIC ympme)
target_compile_definitions(ympme_flight PUBLIC PME_ENABLE_FLIGHT=1
 PME_FLIGHT_ROTORS=$<BOOL:${YMPME_FLIGHT_ROTORS}>
 PME_FLIGHT_AERODYNAMICS=$<BOOL:${YMPME_FLIGHT_AERODYNAMICS}>)
if(YMPME_FLIGHT_ROTORS)
 target_sources(ympme_flight PRIVATE YMPME_Rotor.c)
endif()
if(YMPME_FLIGHT_AERODYNAMICS)
 target_sources(ympme_flight PRIVATE YMPME_Aerodynamics.c)
endif()
if(NOT MSVC)
 target_compile_options(ympme_flight PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
